using System.Globalization;
using System.Text.Json;
using System.Text.Json.Serialization;

static string Ser<T>(T v, JsonSerializerOptions? o = null) => JsonSerializer.Serialize(v, o);
static string Try(Func<string> f)
{
    try { return f(); }
    catch (Exception e) { return "ERR " + e.GetType().Name + ": " + (e.Message.Length > 100 ? e.Message[..100] : e.Message); }
}
static void P(string label, string value) => Console.WriteLine($"{label,-58} => {value}");

Console.WriteLine("--- double formatting thresholds ---");
foreach (var d in new[] { 1e14, 1e15, 1.5e15, 1e16, 1.5e16, 9.9e16, 1e17, 1.2345678901234568e17, 123456789012345678.0, 1e20, 0.001, 0.0001, 0.00001, 0.000123, 0.0000123, 1.5e-5, 1234.5678e10, 2.5e-5, 1e-4, 100000000000000000000.0 })
    P("ser double " + d.ToString("R", CultureInfo.InvariantCulture), Ser(d));
foreach (var f in new[] { 1e14f, 1e15f, 1e16f, 1e7f, 1.5e7f, 1e8f, 123456.7f, 1e-4f, 1e-5f, 1.5e-5f })
    P("ser float " + f.ToString("R", CultureInfo.InvariantCulture), Ser(f));
